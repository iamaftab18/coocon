# Cocoon sorter - Raspberry Pi 4B (Raspberry Pi OS, 64-bit recommended)
#
#   python3 -m venv --system-site-packages ~/cocoon-env     (see README, Part D)
#   source ~/cocoon-env/bin/activate
#   pip install -r requirements.txt                          (about a minute - no PyTorch)
#
# The model runs with NCNN. PyTorch is deliberately NOT used on the Pi: its prebuilt wheels crash
# with "Illegal instruction" on a Raspberry Pi 4 (see README, Troubleshooting).
# Works the same on a Windows / Linux PC (GPIO is simulated there).

numpy
opencv-python           # on the Pi you may use the apt version instead: sudo apt install python3-opencv
ncnn                    # runs the trained model (best_ncnn_model/)
pyyaml                  # reads best_ncnn_model/metadata.yaml
gpiozero>=2.0           # push button, relay, L298N motors
